   309→function FadeIn({ children, delay = 0, className = '', direction = 'up' }: {
   310→  children: React.ReactNode;
   311→  delay?: number;
   312→  className?: string;
   313→  direction?: 'up' | 'down' | 'left' | 'right';
   314→}) {
   315→  const ref = useRef(null);
   316→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   317→
   318→  const dirs: Record<string, { y: number; x: number }> = {
   319→    up: { y: 50, x: 0 },
   320→    down: { y: -50, x: 0 },
   321→    left: { y: 0, x: 50 },
   322→    right: { y: 0, x: -50 },
   323→  };
   324→
   325→  return (
   326→    <motion.div
   327→      ref={ref}
   328→      className={className}
   329→      initial={{ opacity: 0, ...dirs[direction] }}
   330→      animate={isInView ? { opacity: 1, y: 0, x: 0 } : {}}
   331→      transition={{ duration: 0.8, delay, ease: [0.25, 0.46, 0.45, 0.94] }}
   332→    >
   333→      {children}
   334→    </motion.div>
   335→  );
   336→}

   338→function ParallaxImg({ src, alt, className = '', speed = 0.3 }: {
   339→  src: string;
   340→  alt: string;
   341→  className?: string;
   342→  speed?: number;
   343→}) {
   344→  const ref = useRef(null);
   345→  const { scrollYProgress } = useScroll({ target: ref, offset: ['start end', 'end start'] });
   346→  const y = useTransform(scrollYProgress, [0, 1], ['-10%', '10%']);
   347→  const scale = useTransform(scrollYProgress, [0, 0.5, 1], [1.15, 1, 1.15]);
   348→
   349→  return (
   350→    <motion.div ref={ref} className={`overflow-hidden ${className}`}>
   351→      <motion.img
   352→        src={src}
   353→        alt={alt}
   354→        className="w-full h-full object-cover"
   355→        style={{ y, scale }}
   356→        loading="lazy"
   357→      />
   358→    </motion.div>
   359→  );
   360→}

   391→function AnimatedCounter({ target, suffix }: { target: number; suffix: string }) {
   392→  const [count, setCount] = useState(0);
   393→  const ref = useRef<HTMLSpanElement>(null);
   394→  const hasAnimated = useRef(false);
   395→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   396→
   397→  useEffect(() => {
   398→    if (!isInView || hasAnimated.current) return;
   399→    hasAnimated.current = true;
   400→
   401→    const duration = 2000;
   402→    const startTime = Date.now();
   403→
   404→    const step = () => {
   405→      const elapsed = Date.now() - startTime;
   406→      const progress = Math.min(elapsed / duration, 1);
   407→      // Ease out cubic
   408→      const eased = 1 - Math.pow(1 - progress, 3);
   409→      setCount(Math.round(eased * target));
   410→      if (progress < 1) {
   411→        requestAnimationFrame(step);
   412→      }
   413→    };
   414→    requestAnimationFrame(step);
   415→  }, [isInView, target]);
   416→
   417→  return (
   418→    <span ref={ref} className="tabular-nums">
   419→      {count}{suffix}
   420→    </span>
   421→  );
   422→}
