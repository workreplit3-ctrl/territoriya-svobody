   310→function FadeIn({ children, delay = 0, className = '', direction = 'up' }: {
   311→  children: React.ReactNode;
   312→  delay?: number;
   313→  className?: string;
   314→  direction?: 'up' | 'down' | 'left' | 'right';
   315→}) {
   316→  const ref = useRef(null);
   317→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   318→
   319→  const dirs: Record<string, { y: number; x: number }> = {
   320→    up: { y: 50, x: 0 },
   321→    down: { y: -50, x: 0 },
   322→    left: { y: 0, x: 50 },
   323→    right: { y: 0, x: -50 },
   324→  };
   325→
   326→  return (
   327→    <motion.div
   328→      ref={ref}
   329→      className={className}
   330→      initial={{ opacity: 0, ...dirs[direction] }}
   331→      animate={isInView ? { opacity: 1, y: 0, x: 0 } : {}}
   332→      transition={{ duration: 0.8, delay, ease: [0.25, 0.46, 0.45, 0.94] }}
   333→    >
   334→      {children}
   335→    </motion.div>
   336→  );
   337→}

   339→function ParallaxImg({ src, alt, className = '', speed = 0.3 }: {
   340→  src: string;
   341→  alt: string;
   342→  className?: string;
   343→  speed?: number;
   344→}) {
   345→  const ref = useRef(null);
   346→  const { scrollYProgress } = useScroll({ target: ref, offset: ['start end', 'end start'] });
   347→  const y = useTransform(scrollYProgress, [0, 1], ['-10%', '10%']);
   348→  const scale = useTransform(scrollYProgress, [0, 0.5, 1], [1.15, 1, 1.15]);
   349→
   350→  return (
   351→    <motion.div ref={ref} className={`overflow-hidden ${className}`}>
   352→      <motion.img
   353→        src={src}
   354→        alt={alt}
   355→        className="w-full h-full object-cover"
   356→        style={{ y, scale }}
   357→        loading="lazy"
   358→      />
   359→    </motion.div>
   360→  );
   361→}

   392→function AnimatedCounter({ target, suffix }: { target: number; suffix: string }) {
   393→  const [count, setCount] = useState(0);
   394→  const ref = useRef<HTMLSpanElement>(null);
   395→  const hasAnimated = useRef(false);
   396→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   397→
   398→  useEffect(() => {
   399→    if (!isInView || hasAnimated.current) return;
   400→    hasAnimated.current = true;
   401→
   402→    const duration = 2000;
   403→    const startTime = Date.now();
   404→
   405→    const step = () => {
   406→      const elapsed = Date.now() - startTime;
   407→      const progress = Math.min(elapsed / duration, 1);
   408→      // Ease out cubic
   409→      const eased = 1 - Math.pow(1 - progress, 3);
   410→      setCount(Math.round(eased * target));
   411→      if (progress < 1) {
   412→        requestAnimationFrame(step);
   413→      }
   414→    };
   415→    requestAnimationFrame(step);
   416→  }, [isInView, target]);
   417→
   418→  return (
   419→    <span ref={ref} className="tabular-nums">
   420→      {count}{suffix}
   421→    </span>
   422→  );
   423→}
