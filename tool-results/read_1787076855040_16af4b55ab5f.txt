   310→   310→function FadeIn({ children, delay = 0, className = '', direction = 'up' }: {
   311→   311→  children: React.ReactNode;
   312→   312→  delay?: number;
   313→   313→  className?: string;
   314→   314→  direction?: 'up' | 'down' | 'left' | 'right';
   315→   315→}) {
   316→   316→  const ref = useRef(null);
   317→   317→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   318→   318→
   319→   319→  const dirs: Record<string, { y: number; x: number }> = {
   320→   320→    up: { y: 50, x: 0 },
   321→   321→    down: { y: -50, x: 0 },
   322→   322→    left: { y: 0, x: 50 },
   323→   323→    right: { y: 0, x: -50 },
   324→   324→  };
   325→   325→
   326→   326→  return (
   327→   327→    <motion.div
   328→   328→      ref={ref}
   329→   329→      className={className}
   330→   330→      initial={{ opacity: 0, ...dirs[direction] }}
   331→   331→      animate={isInView ? { opacity: 1, y: 0, x: 0 } : {}}
   332→   332→      transition={{ duration: 0.8, delay, ease: [0.25, 0.46, 0.45, 0.94] }}
   333→   333→    >
   334→   334→      {children}
   335→   335→    </motion.div>
   336→   336→  );
   337→   337→}

   339→   339→function ParallaxImg({ src, alt, className = '', speed = 0.3 }: {

   392→   392→function AnimatedCounter({ target, suffix }: { target: number; suffix: string }) {
   393→   393→  const [count, setCount] = useState(0);
   394→   394→  const ref = useRef<HTMLSpanElement>(null);
   395→   395→  const hasAnimated = useRef(false);
   396→   396→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   397→   397→
   398→   398→  useEffect(() => {
   399→   399→    if (!isInView || hasAnimated.current) return;
   400→   400→    hasAnimated.current = true;
   401→   401→
   402→   402→    const duration = 2000;
   403→   403→    const startTime = Date.now();
   404→   404→
   405→   405→    const step = () => {
   406→   406→      const elapsed = Date.now() - startTime;
   407→   407→      const progress = Math.min(elapsed / duration, 1);
   408→   408→      // Ease out cubic
   409→   409→      const eased = 1 - Math.pow(1 - progress, 3);
   410→   410→      setCount(Math.round(eased * target));
   411→   411→      if (progress < 1) {
   412→   412→        requestAnimationFrame(step);
   413→   413→      }
   414→   414→    };
   415→   415→    requestAnimationFrame(step);
   416→   416→  }, [isInView, target]);
   417→   417→
   418→   418→  return (
   419→   419→    <span ref={ref} className="tabular-nums">
   420→   420→      {count}{suffix}
   421→   421→    </span>
   422→   422→  );
   423→   423→}
