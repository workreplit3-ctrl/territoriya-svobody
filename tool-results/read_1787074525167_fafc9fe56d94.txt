    88→function FadeIn({ children, delay = 0, className = '', direction = 'up' }: {
    89→  children: React.ReactNode;
    90→  delay?: number;
    91→  className?: string;
    92→  direction?: 'up' | 'down' | 'left' | 'right';
    93→}) {
    94→  const ref = useRef(null);
    95→  const isInView = useInView(ref, { once: true, margin: '-60px' });
    96→
    97→  const dirs: Record<string, { y: number; x: number }> = {
    98→    up: { y: 50, x: 0 },
    99→    down: { y: -50, x: 0 },
   100→    left: { y: 0, x: 50 },
   101→    right: { y: 0, x: -50 },
   102→  };
   103→
   104→  return (
   105→    <motion.div
   106→      ref={ref}
   107→      className={className}
   108→      initial={{ opacity: 0, ...dirs[direction] }}
   109→      animate={isInView ? { opacity: 1, y: 0, x: 0 } : {}}
   110→      transition={{ duration: 0.8, delay, ease: [0.25, 0.46, 0.45, 0.94] }}
   111→    >
   112→      {children}
   113→    </motion.div>
   114→  );
   115→}

   117→function ParallaxImg({ src, alt, className = '', speed = 0.3 }: {
   118→  src: string;
   119→  alt: string;
   120→  className?: string;
   121→  speed?: number;
   122→}) {
   123→  const ref = useRef(null);
   124→  const { scrollYProgress } = useScroll({ target: ref, offset: ['start end', 'end start'] });
   125→  const y = useTransform(scrollYProgress, [0, 1], ['-10%', '10%']);
   126→  const scale = useTransform(scrollYProgress, [0, 0.5, 1], [1.15, 1, 1.15]);
   127→
   128→  return (
   129→    <motion.div ref={ref} className={`overflow-hidden ${className}`}>
   130→      <motion.img
   131→        src={src}
   132→        alt={alt}
   133→        className="w-full h-full object-cover"
   134→        style={{ y, scale }}
   135→        loading="lazy"
   136→      />
   137→    </motion.div>
   138→  );
   139→}

   141→function StaggerChildren({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   142→  const ref = useRef(null);
   143→  const isInView = useInView(ref, { once: true, margin: '-40px' });
   144→
   145→  return (
   146→    <motion.div
   147→      ref={ref}
   148→      className={className}
   149→      initial="hidden"
   150→      animate={isInView ? 'visible' : 'hidden'}
   151→      variants={{
   152→        hidden: {},
   153→        visible: { transition: { staggerChildren: 0.12 } },
   154→      }}
   155→    >
   156→      {children}
   157→    </motion.div>
   158→  );
   159→}
   160→
   161→const staggerItem = {
   162→  hidden: { opacity: 0, y: 40 },
   163→  visible: { opacity: 1, y: 0, transition: { duration: 0.7, ease: [0.25, 0.46, 0.45, 0.94] } },
   164→};

   168→/* ------------------------------------------------------------------ */
   169→
   170→function AnimatedCounter({ target, suffix }: { target: number; suffix: string }) {
   171→  const [count, setCount] = useState(0);
   172→  const ref = useRef<HTMLSpanElement>(null);
   173→  const hasAnimated = useRef(false);
   174→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   175→
   176→  useEffect(() => {
   177→    if (!isInView || hasAnimated.current) return;
   178→    hasAnimated.current = true;
   179→
   180→    const duration = 2000;
   181→    const startTime = Date.now();
   182→
   183→    const step = () => {
   184→      const elapsed = Date.now() - startTime;
   185→      const progress = Math.min(elapsed / duration, 1);
   186→      // Ease out cubic
   187→      const eased = 1 - Math.pow(1 - progress, 3);
   188→      setCount(Math.round(eased * target));
   189→      if (progress < 1) {
   190→        requestAnimationFrame(step);
   191→      }
   192→    };
   193→    requestAnimationFrame(step);
   194→  }, [isInView, target]);
   195→
   196→  return (
   197→    <span ref={ref} className="tabular-nums">
   198→      {count}{suffix}
   199→    </span>
   200→  );
   201→}
